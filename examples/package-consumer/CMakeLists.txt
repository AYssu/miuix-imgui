cmake_minimum_required(VERSION 3.21)
project(skui-package-consumer LANGUAGES CXX)
if(SKUI_NAVIGATION_ONLY)
    find_package(skui 0.1 CONFIG REQUIRED COMPONENTS navigation)
    add_executable(skui-package-consumer navigation.cpp)
    target_link_libraries(skui-package-consumer PRIVATE skui::navigation)
else()
    # Optional fixture for an application that already imports its own ImGui.
    if(SKUI_HOST_IMGUI_PACKAGE)
        include("${SKUI_HOST_IMGUI_PACKAGE}/hostImGuiTargets.cmake")
        set(SKUI_IMGUI_TARGET host::imgui)
    endif()
    find_package(skui 0.1 CONFIG REQUIRED)
    add_executable(skui-package-consumer ../minimal/main.cpp)
    target_link_libraries(skui-package-consumer PRIVATE skui::ui skui::navigation)
endif()
